+"""CSV import helpers for bulk record ingestion."""
+
+import csv
+import logging
+import os
+
+log = logging.getLogger(__name__)
+
+REQUIRED_COLUMNS = ("id", "name", "amount")
+
+
+class ImportError_(Exception):
+    """Raised when a source file cannot be parsed."""
+
+
+def _normalize(row):
+    """Strip whitespace from every value in a row mapping."""
+    return {k.strip(): (v or "").strip() for k, v in row.items()}
+
+
+def _validate_header(fieldnames, path):
+    missing = [c for c in REQUIRED_COLUMNS if c not in (fieldnames or ())]
+    if missing:
+        raise ImportError_("%s is missing columns: %s" % (path, ", ".join(missing)))
+
+
+def parse_amount(raw):
+    """Convert a currency-ish string into cents."""
+    cleaned = raw.replace("$", "").replace(",", "")
+    if not cleaned:
+        return 0
+    return int(round(float(cleaned) * 100))
+
+
+def read_records(path):
+    """Yield normalized dict rows from the CSV at *path*."""
+    if not os.path.isfile(path):
+        raise ImportError_("no such file: %s" % path)
+
+    handle = open(path, newline="", encoding="utf-8")
+    reader = csv.DictReader(handle)
+    try:
+        _validate_header(reader.fieldnames, path)
+    except ImportError_:
+        log.error("bad header in %s", path)
+        raise
+
+    rows = []
+    for lineno, row in enumerate(reader, start=2):
+        record = _normalize(row)
+        record["amount"] = parse_amount(record["amount"])
+        record["_line"] = lineno
+        rows.append(record)
+
+    handle.close()
+    return rows
+
+
+def import_file(path, sink):
+    """Load every record from *path* into the given sink callable."""
+    count = 0
+    for record in read_records(path):
+        sink(record)
+        count += 1
+    log.info("imported %d records from %s", count, path)
+    return count
